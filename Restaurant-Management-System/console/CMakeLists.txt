cmake_minimum_required(VERSION 3.16)
project(QuanLyNhaHangConsole VERSION 1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_executable(restaurant_app src/main.cpp src/restaurant.cpp)
target_include_directories(restaurant_app PRIVATE include)
add_executable(benchmark src/benchmark.cpp)
target_include_directories(benchmark PRIVATE include)
add_executable(test_algorithms tests/test_algorithms.cpp)
target_include_directories(test_algorithms PRIVATE include)
enable_testing()
add_test(NAME algorithms COMMAND test_algorithms)
